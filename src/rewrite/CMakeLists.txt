# Fresh PCM voice mixing and optional native audio output over the pinned SDK.
add_library(pinball_voice_mixer STATIC voice_mixer.cpp)
target_link_libraries(pinball_voice_mixer PUBLIC pinball_wave)
target_compile_features(pinball_voice_mixer PUBLIC cxx_std_11)
add_library(pinball_xbox_audio_output STATIC xbox_audio_output.cpp)
target_link_libraries(pinball_xbox_audio_output PUBLIC pinball_voice_mixer)
target_compile_features(pinball_xbox_audio_output PUBLIC cxx_std_11)
if(TARGET pinball360)
    target_link_libraries(pinball360 PRIVATE pinball_xbox_audio_output)
endif()
foreach(feature IN ITEMS voice_mixer xbox_audio_output)
    if(BUILD_TESTING AND NOT XSDK)
        add_executable(pinball_${feature}_tests "../../tests/rewrite/${feature}_tests.cpp")
        target_link_libraries(pinball_${feature}_tests PRIVATE pinball_${feature})
        add_test(NAME pinball_${feature}_tests COMMAND pinball_${feature}_tests)
    endif()
    if(XSDK)
        target_compile_options(pinball_${feature} PRIVATE -O1)
        add_executable(pinball360_${feature}_selftest "../../tests/rewrite/${feature}_tests.cpp")
        target_link_libraries(pinball360_${feature}_selftest PRIVATE pinball_${feature})
        target_compile_options(pinball360_${feature}_selftest PRIVATE -O1)
        xsdk_add_xex(pinball360_${feature}_selftest TITLE_ID 0x50494E54 NAME "Pinball Audio Test" VERSION 0.0.15.0)
    endif()
endforeach()
